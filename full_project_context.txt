from django.contrib.auth.models import BaseUserManager

class UserManager(BaseUserManager):
    def create_user(
            self, email: str, first_name: str, last_name: str, password: str = None, 
            **extra_fields
        ) -> "User":
        if not email:
            raise ValueError("The Email field must be set")
        email = self.normalize_email(email).lower()
        user = self.model(
            email=email,
            first_name=first_name,
            last_name=last_name,
            **extra_fields
        )
        user.set_password(password)
        user.save(using=self._db)
        return user

    def create_superuser(
            self, email: str, first_name: str, last_name: str, password: str = None, 
            **extra_fields
        ) -> "User":
        extra_fields.setdefault("is_staff", True)
        extra_fields.setdefault("is_superuser", True)

        if extra_fields.get("is_staff") is not True:
            raise ValueError("Superuser must have is_staff=True.")
        if extra_fields.get("is_superuser") is not True:
            raise ValueError("Superuser must have is_superuser=True.")

        return self.create_user(email, first_name, last_name, password, **extra_fields)

def dashboard_callback(request: HttpRequest, context: dict) -> dict:
    post_stats = Post.objects.aggregate(
        total=Count("id"),
        published=Count("id", filter=Q(status=Post.StatusChoices.PUBLISHED)),
        drafts=Count("id", filter=Q(status=Post.StatusChoices.DRAFT)),
    )
    total_posts = post_stats["total"]

    categories = Category.objects.annotate(post_count=Count("posts")).order_by("-post_count")[:TOP_CATEGORIES_LIMIT]
    top_authors = User.objects.annotate(post_count=Count("posts")).order_by("-post_count")[:TOP_AUTHORS_LIMIT]

    if total_posts > 0:
        published_percentage = int(post_stats["published"] * PERCENT_MULTIPLIER / total_posts)
        draft_percentage = int(post_stats["drafts"] * PERCENT_MULTIPLIER / total_posts)
    else:
        published_percentage = 0
        draft_percentage = 0

    context.update({
        "total_users": User.objects.count(),
        "published_posts": post_stats["published"],
        "draft_posts": post_stats["drafts"],
        "total_comments": Comment.objects.count(),
        "published_percentage": published_percentage,
        "draft_percentage": draft_percentage,
        "category_chart": json.dumps({
            "labels": [category.name for category in categories],
            "datasets": [{
                "label": "Posts count",
                "data": [category.post_count for category in categories],
                "backgroundColor": CHART_BAR_COLOR,
            }],
        }),
        "top_authors_table": {
            "headers": ["Author Email", "Full Name", "Posts Count"],
            "rows": [
                [author.email, f"{author.first_name} {author.last_name}", author.post_count]
                for author in top_authors
            ],
        },
    })
    return context

class Command(BaseCommand):
    help = "Fill the database with fake data."

    def add_arguments(self, parser: CommandParser) -> None:
        parser.add_argument("--users", type=int, default=DEFAULT_USERS, help="Number of users to create")
        parser.add_argument("--posts", type=int, default=DEFAULT_POSTS, help="Number of posts to create")
        parser.add_argument("--comments", type=int, default=DEFAULT_COMMENTS, help="Number of comments to create")
        parser.add_argument("--clear", action="store_true", help="Delete existing data before seeding")

    def handle(self, *args: Any, **options: Any) -> None:
        users_count: int = options["users"]
        posts_count: int = options["posts"]
        comments_count: int = options["comments"]
        clear: bool = options["clear"]

        if users_count < 0 or posts_count < 0 or comments_count < 0:
            raise CommandError("Numbers of entities cannot be negative.")

        if clear:
            remaining_users = User.objects.filter(is_superuser=True).count()
            remaining_published = 0
        else:
            remaining_users = User.objects.count()
            remaining_published = Post.objects.filter(status=Post.StatusChoices.PUBLISHED).count()

        if (posts_count > 0 or comments_count > 0) and remaining_users + users_count == 0:
            raise CommandError("Cannot create posts or comments because there are no users in the database.")
        if comments_count > 0 and posts_count == 0 and remaining_published == 0:
            raise CommandError("Cannot create comments because there are no published posts.")

        start_time = time.time()

        with transaction.atomic():
            if clear:
                self.stdout.write(self.style.WARNING("Clearing database..."))
                Comment.objects.all().delete()
                Post.objects.all().delete()
                Tag.objects.all().delete()
                Category.objects.all().delete()
                User.objects.filter(is_superuser=False).delete()
                self.stdout.write(self.style.SUCCESS("Database cleared successfully!"))

            users_created = self._create_users(users_count)
            categories_created = self._create_categories()
            tags_created = self._create_tags()
            posts_created = self._create_posts(posts_count, comments_count > 0, remaining_published > 0)
            comments_created = self._create_comments(comments_count)

        elapsed_time = round(time.time() - start_time, ELAPSED_DIGITS)

        self._write_summary("Users", users_created)
        self._write_summary("Categories", categories_created)
        self._write_summary("Tags", tags_created)
        self._write_summary("Posts", posts_created)
        self._write_summary("Comments", comments_created)
        self.stdout.write(self.style.SUCCESS(f"Done in {elapsed_time} seconds"))

    def _write_summary(self, label: str, count: int) -> None:
        self.stdout.write(f"{label + ':':<{SUMMARY_LABEL_WIDTH}}{count:>{SUMMARY_COUNT_WIDTH}} created")

    def _create_users(self, count: int) -> int:
        if count == 0:
            return 0

        used_emails = set(User.objects.values_list("email", flat=True))
        hashed_password = make_password(SEED_PASSWORD)
        users_to_create = []

        for _ in range(count):
            first_name = fake.first_name()
            last_name = fake.last_name()
            base_local = f"{slugify(first_name)}_{slugify(last_name)}"
            domain = fake.free_email_domain()
            email = f"{base_local}@{domain}"
            while email in used_emails:
                suffix = random.randint(EMAIL_SUFFIX_MIN, EMAIL_SUFFIX_MAX)
                email = f"{base_local}{suffix}@{domain}"
            used_emails.add(email)

            users_to_create.append(User(
                email=email,
                first_name=first_name,
                last_name=last_name,
                password=hashed_password,
                is_active=random.random() >= INACTIVE_USER_RATIO,
                is_staff=random.random() < STAFF_USER_RATIO,
            ))

        return len(User.objects.bulk_create(users_to_create))

    def _create_categories(self) -> int:
        existing_slugs = set(Category.objects.values_list("slug", flat=True))
        categories_to_create = [

        user_ids = list(User.objects.values_list("id", flat=True))

        tag_ids = list(Tag.objects.values_list("id", flat=True))

        posts_to_create = []
        for _ in range(count):
            title = fake.sentence(nb_words=random.randint(MIN_TITLE_WORDS, MAX_TITLE_WORDS)).rstrip(".")
            base_slug = slugify(title) or DEFAULT_SLUG
            slug = base_slug
            suffix = FIRST_SLUG_SUFFIX
            while slug in used_slugs:
                slug = f"{base_slug}-{suffix}"
                suffix += 1
            used_slugs.add(slug)

            is_published = random.random() < PUBLISHED_POST_RATIO
            category_id = None if random.random() < NO_CATEGORY_RATIO else random.choice(category_ids)

            posts_to_create.append(Post(
                author_id=random.choice(user_ids),
                title=title,
                slug=slug,
                body="\n\n".join(fake.paragraphs(nb=random.randint(MIN_PARAGRAPHS, MAX_PARAGRAPHS))),
                category_id=category_id,
                status=Post.StatusChoices.PUBLISHED if is_published else Post.StatusChoices.DRAFT,
            ))

        if need_published and not has_published:
            if not any(post.status == Post.StatusChoices.PUBLISHED for post in posts_to_create):
                posts_to_create[0].status = Post.StatusChoices.PUBLISHED

        created_posts = Post.objects.bulk_create(posts_to_create)

        post_tag_model = Post.tags.through
        relations = []
        for post in created_posts:
            tags_amount = min(random.randint(MIN_TAGS_PER_POST, MAX_TAGS_PER_POST), len(tag_ids))
            for tag_id in random.sample(tag_ids, k=tags_amount):
                relations.append(post_tag_model(post_id=post.id, tag_id=tag_id))
        if relations:
            post_tag_model.objects.bulk_create(relations)

        return len(created_posts)

    def _create_comments(self, count: int) -> int:
        if count == 0:
            return 0

        published_ids = list(
            Post.objects.filter(status=Post.StatusChoices.PUBLISHED).values_list("id", flat=True)
        )
        if not published_ids:
            raise CommandError("Cannot create comments because there are no published posts.")

        user_ids = list(User.objects.values_list("id", flat=True))
        comments_to_create = [
            Comment(
                post_id=random.choice(published_ids),
                author_id=random.choice(user_ids),
                body=" ".join(fake.sentences(nb=random.randint(MIN_COMMENT_SENTENCES, MAX_COMMENT_SENTENCES))),
            )
            for _ in range(count)
        ]
        return len(Comment.objects.bulk_create(comments_to_create))

# Django Blog API & Advanced Admin Panel

A modern web application built with Django 5.x, featuring a comprehensive blog platform with a highly optimized custom development data seeder and a customized, responsive administration panel powered by the **Django Unfold** theme (Tailwind CSS).

---

## 🛠 Tech Stack

* **Backend:** Python 3.11+, Django 5.x
* **Database:** PostgreSQL / SQLite3

## 🚀 Local Deployment

Follow these exact steps to set up and run the project locally:

### 1. Clone the Repository
```bash
git clone <your-repository-url>
cd <project-folder-name>
```

### 2. Configure the Virtual Environment
```bash
python -m venv venv
source venv/bin/activate  # For Linux/macOS
# or
venv\Scripts\activate     # For Windows
```

### 3. Install Dependencies
```bash
pip install -r requirements/dev.txt
```

### 4. Setup Environment Variables
Create a `.env` file in the root directory of the project using the following template:
```env
BLOG_SECRET_KEY=your-super-secret-django-key
BLOG_ENV_ID=local
```

### 5. Apply Database Migrations
```bash
python manage.py migrate
```

### 6. Run the Data Seeder (Seed Command)
To populate the database with optimized mock data for development, execute the custom management command:
```bash
python manage.py seed --users 50 --posts 300 --comments 1500
```

**Available Command-Line Arguments:**
* `--users` (Default: `20`): Number of users to create.
* `--posts` (Default: `100`): Number of posts to create.
* `--comments` (Default: `500`): Number of comments to create.
* `--clear`: Optional flag. When passed, it deletes existing comments, posts, tags, categories, and all users **except superusers** before seeding.

### 7. Create Superuser & Start Server
```bash
python manage.py createsuperuser
python manage.py runserver
```
Once started, the administration panel will be available at: `http://127.0.0`

## ⚡ Seed Command Architecture & Optimization (`seed.py`)

The custom data generation command is engineered for maximum performance, minimizing database overhead and preventing crashes on consecutive runs:
* **Batch Insertions (`bulk_create`):** All records for users, categories, tags, posts, and comments are inserted into the database using single batch operations, reducing thousands of SQL queries down to a handful.
* **No N+1 M2M Insertion:** Rather than resolving Many-to-Many relationships inside a loop, tags are bound by performing a bulk insertion on the intermediate join table (`Post.tags.through`), satisfying strict performance requirements.
* **Proportional Generation Logic:** Adheres strictly to predefined distribution requirements: approximately 70% of posts are generated as `published`, 30% as `draft`, and around 10% are created without a category assigned.
* **In-Memory Cache Filtering (`set`):** Uniqueness checks for emails and slugs are handled entirely via Python `set` structures in RAM, eliminating redundant database query hits.
* **Single-Pass Password Hashing:** User passwords are encrypted exactly once using `make_password(SEED_PASSWORD)` before execution, avoiding massive performance bottlenecks.
* **Rigorous Validation:** Throws explicit `CommandError` instances if negative inputs are provided or if dependent relationships are missing (e.g., generating posts when there are no users).

Console output layout summary example:
```text

Done in 2.41 seconds
```
